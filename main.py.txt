from fastapi import FastAPI, HTTPException
from pydantic import BaseModel, HttpUrl
import requests
import pandas as pd
from typing import List, Dict, Any

app = FastAPI(
    title="HTML Table to JSON Parser for AI Agents",
    description="Extracts tables from any webpage and converts them to clean JSON for LLMs.",
    version="1.0.0"
)

class TableExtractRequest(BaseModel):
    url: HttpUrl

class TableExtractResponse(BaseModel):
    url: str
    tables_found: int
    tables: List[List[Dict[str, Any]]]

@app.post("/api/v1/extract-tables", response_model=TableExtractResponse)
def extract_tables(request: TableExtractRequest):
    headers = {
        "User-Agent": "Mozilla/5.0 (Windows NT 10.0; Win64; x64) AppleWebKit/537.36 (KHTML, like Gecko) Chrome/120.0.0.0 Safari/537.36"
    }
    try:
        response = requests.get(str(request.url), headers=headers, timeout=10)
        response.raise_for_status()
    except Exception as e:
        raise HTTPException(status_code=400, detail=f"Failed to fetch URL: {str(e)}")

    try:
        dfs = pd.read_html(response.text)
        extracted_tables = []
        for df in dfs:
            cleaned_df = df.fillna("")
            extracted_tables.append(cleaned_df.to_dict(orient="records"))

        return TableExtractResponse(
            url=str(request.url),
            tables_found=len(extracted_tables),
            tables=extracted_tables
        )
    except Exception as e:
        return TableExtractResponse(
            url=str(request.url),
            tables_found=0,
            tables=[]
        )

if __name__ == "__main__":
    import uvicorn
    uvicorn.run("main:app", host="0.0.0.0", port=8000)
